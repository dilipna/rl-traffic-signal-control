import numpy as np
import matplotlib.pyplot as plt
from io import BytesIO
import base64

class TrafficEnv:
    """
    Simple 2-road intersection:
    - NS direction queue
    - EW direction queue
    Traffic light phase:
    - phase = 0 : NS green
    - phase = 1 : EW green

    Arrivals: Poisson(lambda_ns), Poisson(lambda_ew)
    Departures: if green, discharge up to 'service' cars per step
    """
    def __init__(self, lambda_ns=2.0, lambda_ew=2.0, service=3, max_q=50, seed=0):
        self.lambda_ns = float(lambda_ns)
        self.lambda_ew = float(lambda_ew)
        self.service = int(service)
        self.max_q = int(max_q)
        self.rng = np.random.default_rng(seed)
        self.reset()

    def reset(self):
        self.q_ns = 0
        self.q_ew = 0
        self.phase = 0  # start with NS green
        return self._state()

    def _bin(self, q):
        # Discretize queue into 5 bins: 0..4
        # 0: 0-2, 1: 3-6, 2: 7-12, 3: 13-20, 4: 21+
        if q <= 2: return 0
        if q <= 6: return 1
        if q <= 12: return 2
        if q <= 20: return 3
        return 4

    def _state(self):
        return (self._bin(self.q_ns), self._bin(self.q_ew), self.phase)

    def step(self, action):
        """
        action:
        0 = keep current phase
        1 = switch phase
        """
        if action == 1:
            self.phase = 1 - self.phase

        # arrivals
        a_ns = self.rng.poisson(self.lambda_ns)
        a_ew = self.rng.poisson(self.lambda_ew)

        self.q_ns = min(self.max_q, self.q_ns + a_ns)
        self.q_ew = min(self.max_q, self.q_ew + a_ew)

        # departures
        if self.phase == 0:  # NS green
            d = min(self.service, self.q_ns)
            self.q_ns -= d
        else:               # EW green
            d = min(self.service, self.q_ew)
            self.q_ew -= d

        # reward = negative total queue (minimize congestion)
        reward = -(self.q_ns + self.q_ew)

        return self._state(), reward

def train_q_learning(lambda_ns=2.0, lambda_ew=2.0, episodes=200, steps=60, seed=0,
                     alpha=0.15, gamma=0.95, eps_start=1.0, eps_end=0.05):
    env = TrafficEnv(lambda_ns=lambda_ns, lambda_ew=lambda_ew, seed=seed)
    # state space: 5 x 5 x 2 = 50, actions = 2
    Q = np.zeros((5, 5, 2, 2), dtype=np.float32)

    rewards_per_ep = []

    for ep in range(int(episodes)):
        s = env.reset()
        eps = eps_end + (eps_start - eps_end) * np.exp(-ep / max(1, episodes/5))
        total = 0

        for _ in range(int(steps)):
            # epsilon-greedy
            if env.rng.random() < eps:
                a = env.rng.integers(0, 2)
            else:
                a = int(np.argmax(Q[s[0], s[1], s[2], :]))

            s2, r = env.step(a)
            total += r

            # Q update
            best_next = np.max(Q[s2[0], s2[1], s2[2], :])
            Q[s[0], s[1], s[2], a] = (1 - alpha) * Q[s[0], s[1], s[2], a] + alpha * (r + gamma * best_next)

            s = s2

        rewards_per_ep.append(total)

    return Q, rewards_per_ep

def eval_policy(Q, lambda_ns=2.0, lambda_ew=2.0, steps=120, seed=123):
    env = TrafficEnv(lambda_ns=lambda_ns, lambda_ew=lambda_ew, seed=seed)
    s = env.reset()
    total_queue = 0

    for _ in range(int(steps)):
        a = int(np.argmax(Q[s[0], s[1], s[2], :]))
        s, _ = env.step(a)
        total_queue += (env.q_ns + env.q_ew)

    avg_queue = total_queue / steps
    return avg_queue

def eval_baseline(lambda_ns=2.0, lambda_ew=2.0, steps=120, seed=123):
    """
    Baseline: fixed-time switching every 10 steps
    """
    env = TrafficEnv(lambda_ns=lambda_ns, lambda_ew=lambda_ew, seed=seed)
    env.reset()
    total_queue = 0

    for t in range(int(steps)):
        # switch every 10 steps
        action = 1 if (t % 10 == 0 and t != 0) else 0
        _, _ = env.step(action)
        total_queue += (env.q_ns + env.q_ew)

    avg_queue = total_queue / steps
    return avg_queue

lambda_ns, lambda_ew = 2.5, 2.5
Q, rewards = train_q_learning(lambda_ns=lambda_ns, lambda_ew=lambda_ew, episodes=200, steps=60, seed=0)

baseline = eval_baseline(lambda_ns=lambda_ns, lambda_ew=lambda_ew)
trained  = eval_policy(Q, lambda_ns=lambda_ns, lambda_ew=lambda_ew)

print("Baseline avg queue:", round(baseline, 2))
print("RL policy avg queue:", round(trained, 2))

plt.figure()
plt.plot(rewards)
plt.title("Training Reward per Episode (higher is better / less negative)")
plt.xlabel("Episode")
plt.ylabel("Total Reward")
plt.show()

def fig_to_base64(fig):
    buf = BytesIO()
    fig.savefig(buf, format="png", bbox_inches="tight", dpi=140)
    plt.close(fig)
    buf.seek(0)
    return base64.b64encode(buf.read()).decode("utf-8")

@app.route("/", methods=["GET", "POST"])

    steps     = request.form.get("steps", "60")

    baseline = trained = ""

        plt.ylabel("Total Reward")
